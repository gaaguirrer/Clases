Script auditoría de servidor

USE [master]

GO

CREATE SERVER AUDIT [Adit_Server_Level]
TO FILE 
(	FILEPATH = N'F:\f'
	,MAXSIZE = 0 MB
	,MAX_ROLLOVER_FILES = 2147483647
	,RESERVE_DISK_SPACE = OFF
)
WITH
(	QUEUE_DELAY = 1000
	,ON_FAILURE = CONTINUE
)

GO

Script auditoría sql

USE [master]

GO

CREATE SERVER AUDIT SPECIFICATION [Audit_Sql_Level]
FOR SERVER AUDIT [Adit_Server_Level]
ADD (DATABASE_ROLE_MEMBER_CHANGE_GROUP),
ADD (SERVER_ROLE_MEMBER_CHANGE_GROUP),
ADD (DATABASE_PERMISSION_CHANGE_GROUP),
ADD (SCHEMA_OBJECT_PERMISSION_CHANGE_GROUP),
ADD (SERVER_PERMISSION_CHANGE_GROUP),
ADD (DATABASE_PRINCIPAL_CHANGE_GROUP),
ADD (DATABASE_PERMISSION_CHANGE_GROUP)

GO



Script auditoría de base de datos

USE [Verdulero]

GO

CREATE DATABASE AUDIT SPECIFICATION [Audit_BD_Level]
FOR SERVER AUDIT [Adit_Server_Level]
ADD (SELECT ON DATABASE::[Verdulero] BY [public]),
ADD (UPDATE ON DATABASE::[Verdulero] BY [public]),
ADD (DELETE ON DATABASE::[Verdulero] BY [public])

GO
