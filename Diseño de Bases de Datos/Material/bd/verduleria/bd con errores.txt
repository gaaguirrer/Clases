-- Crear la base de datos
CREATE DATABASE verdulería;

-- Usar la base de datos
USE verdulería;

-- Crear tabla Grupos
CREATE TABLE Grupos (
  id_grupo INT PRIMARY KEY,
  nombre_grupo VARCHAR(100)
);

-- Crear tabla Productos
CREATE TABLE Productos (
  id_producto INT PRIMARY KEY,
  nombre VARCHAR(100),
  id_grupo INT,
  precio DECIMAL(10, 2),
  FOREIGN KEY (id_grupo) REFERENCES Grupos(id_grupo)
);

-- Crear tabla Vendedores
CREATE TABLE Vendedores (
  id_vendedor INT PRIMARY KEY,
  nombre VARCHAR(100),
  apellido VARCHAR(100),
  fecha_alta DATE,
  nif VARCHAR(20),
  fecha_nacimiento DATE,
  direccion VARCHAR(200),
  municipio VARCHAR(100),
  telefono VARCHAR(20),
  estado_civil VARCHAR(20)
);

-- Crear tabla Ventas
CREATE TABLE Ventas (
  cod_vendedor INT,
  cod_producto INT,
  fecha DATE,
  kilos DECIMAL(10, 2),
  FOREIGN KEY (cod_vendedor) REFERENCES Vendedores(id_vendedor),
  FOREIGN KEY (cod_producto) REFERENCES Productos(id_producto)
);


-- Crear tabla Ventas
CREATE TABLE Ventas (
  cod_vendedor INT,
  cod_producto INT,
  fecha DATE,
  kilos DECIMAL(10, 2),
  FOREIGN KEY (cod_vendedor) REFERENCES Vendedores(id_vendedor),
  FOREIGN KEY (cod_producto) REFERENCES Productos(id_producto)
);
