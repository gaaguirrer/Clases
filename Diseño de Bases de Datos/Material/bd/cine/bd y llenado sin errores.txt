-- Crear la base de datos
CREATE DATABASE CineDB;

-- Usar la base de datos
USE CineDB;

-- Crear tabla Peliculas
CREATE TABLE Peliculas (
  PeliculaID INT PRIMARY KEY,
  Titulo VARCHAR(100),
  Duracion INT,
  GeneroID INT,
  FOREIGN KEY (GeneroID) REFERENCES Generos(GeneroID)
);

-- Crear tabla Salas
CREATE TABLE Salas (
  SalaID INT PRIMARY KEY,
  Nombre VARCHAR(100),
  Capacidad INT
);

-- Crear tabla Horarios
CREATE TABLE Horarios (
  HorarioID INT PRIMARY KEY,
  Hora TIME,
  Fecha DATE
);

-- Crear tabla Generos
CREATE TABLE Generos (
  GeneroID INT PRIMARY KEY,
  Nombre VARCHAR(100)
);

-- Crear tabla Boletos
CREATE TABLE Boletos (
  BoletoID INT PRIMARY KEY,
  PeliculaID INT,
  SalaID INT,
  HorarioID INT,
  Asiento VARCHAR(10),
  Precio DECIMAL(8, 2),
  FOREIGN KEY (PeliculaID) REFERENCES Peliculas(PeliculaID),
  FOREIGN KEY (SalaID) REFERENCES Salas(SalaID),
  FOREIGN KEY (HorarioID) REFERENCES Horarios(HorarioID)
);
